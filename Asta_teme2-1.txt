local b64chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
local b64lookup = {}
for i = 1, 64 do b64lookup[string.byte(b64chars, i)] = i - 1 end

local function decodeBase64(data)
    data = data:gsub("[^%w%+/=]", "")
    local nativeDecode = (crypt and (crypt.base64decode or (crypt.base64 and crypt.base64.decode)))
        or (syn and syn.crypt and syn.crypt.base64 and syn.crypt.base64.decode)
        or base64_decode or (base64 and base64.decode)
    if nativeDecode then
        local ok, res = pcall(nativeDecode, data)
        if ok and type(res) == "string" and #res > 1000 then return res end
    end
    local out = {}
    local count = 0
    for i = 1, #data, 4 do
        local a, b, c, d = string.byte(data, i, i + 3)
        a, b = b64lookup[a], b64lookup[b]
        c = c and b64lookup[c]
        d = d and b64lookup[d]
        local n = a * 262144 + b * 4096 + (c or 0) * 64 + (d or 0)
        local b1 = math.floor(n / 65536)
        local b2 = math.floor(n / 256) % 256
        local b3 = n % 256
        if c and d then
            out[#out + 1] = string.char(b1, b2, b3)
        elseif c then
            out[#out + 1] = string.char(b1, b2)
        else
            out[#out + 1] = string.char(b1)
        end
        count = count + 1
        if count % 15000 == 0 then task.wait() end
    end
    return table.concat(out)
end

local MOSAIC_DATA = {}
local MOSAIC_SEGS, MOSAIC_PER, MOSAIC_ROWS = 5, 20, 72
local mosaicTemplate = nil
local function makeTemplate()
    local holder = Instance.new("Frame")
    holder.Name = "AstaMosaic"
    holder.Size = UDim2.new(1, 0, 1, 0)
    holder.Position = UDim2.new(0, 0, 0, 0)
    holder.BackgroundColor3 = Color3.fromRGB(40, 20, 80)
    holder.BorderSizePixel = 0
    holder.ClipsDescendants = true
    holder.ZIndex = 0
    for r = 0, MOSAIC_ROWS - 1 do
        local rowStr = MOSAIC_DATA[r + 1]
        for k = 0, MOSAIC_SEGS - 1 do
            local kp = {}
            for i = 0, MOSAIC_PER - 1 do
                local p = (k * (MOSAIC_PER - 1) + i) * 6 + 1
                local cr = tonumber(string.sub(rowStr, p, p + 1), 16)
                local cg = tonumber(string.sub(rowStr, p + 2, p + 3), 16)
                local cb = tonumber(string.sub(rowStr, p + 4, p + 5), 16)
                kp[#kp + 1] = ColorSequenceKeypoint.new(i / (MOSAIC_PER - 1), Color3.fromRGB(cr, cg, cb))
            end
            local cell = Instance.new("Frame")
            cell.BorderSizePixel = 0
            cell.BackgroundColor3 = Color3.new(1, 1, 1)
            cell.Size = UDim2.new(1 / MOSAIC_SEGS, 1, 1 / MOSAIC_ROWS, 1)
            cell.Position = UDim2.new(k / MOSAIC_SEGS, 0, r / MOSAIC_ROWS, 0)
            cell.ZIndex = 0
            local gr = Instance.new("UIGradient")
            gr.Color = ColorSequence.new(kp)
            gr.Parent = cell
            cell.Parent = holder
        end
    end
    return holder
end

end

runService.Heartbeat:Connect(function()
    if not autoActive then return end

    local character = localPlayer.Character
    if not character then return end

    local hasTool = false
    for _, item in ipairs(character:GetChildren()) do
        if item:IsA("Tool") then
            hasTool = true
            break
        end
    end
    if not hasTool then return end

    local humanoid = character:FindFirstChild("Humanoid")
    local rootPart = character:FindFirstChild("HumanoidRootPart")
    if not humanoid or not rootPart then return end

    local nearestTarget = nil
    local shortestDist = math.huge

    for _, p in ipairs(players:GetPlayers()) do
        if p ~= localPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            local targetHumanoid = p.Character:FindFirstChild("Humanoid")
            if targetHumanoid and targetHumanoid.Health > 0 then
                local dist = (rootPart.Position - p.Character.HumanoidRootPart.Position).Magnitude
                if dist < shortestDist then
                    shortestDist = dist
                    nearestTarget = p.Character.HumanoidRootPart
                end
            end
        end
    end

    if nearestTarget then
        humanoid:MoveTo(nearestTarget.Position)
    end
end)
