local function tween(obj, time, props, style, dir)
  local tw = TweenService:Create(obj, TweenInfo.new(time, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out), props)

local function hwidFromGetHWID()
  return type(getHWID) == "function" and getHWID()
end
local function hwidFromGethwid()
  return type(gethwid) == "function" and gethwid()
end
local function hwidFromSyn()
  return (syn and type(syn.gethwid) == "function") and syn.gethwid()
end
local function hwidFromGet_hwid()
  return type(get_hwid) == "function" and get_hwid()
end

local function getHwid()
  for _, fn in ipairs({hwidFromGethwid, hwidFromGet_hwid, hwidFromGetHWID, hwidFromSyn}) do
    local ok, value = pcall(fn)
    if ok and type(value) == "string" then
      local clean = value:gsub("%s+", "")
      if #clean >= 8 and #clean <= 256 then
        return clean
      end
    end
  end
  return nil
end

local function formatDuration(seconds)
  local total = math.max(0, math.floor(seconds))
  local days = math.floor(total / 86400)
  local hours = math.floor(total % 86400 / 3600)
  local minutes = math.floor(total % 3600 / 60)
  local secs = total % 60
  if days > 0 then
    return string.format("%dd %02dh %02dm", days, hours, minutes)
  end
  if hours > 0 then
    return string.format("%dh %02dm %02ds", hours, minutes, secs)
  end
  return string.format("%02dm %02ds", minutes, secs)
end

local function parseIsoTimestamp(isoString)
  if type(isoString) ~= "string" then
    return nil
  end
  local ok, dt = pcall(function()
    return DateTime.fromIsoDate(isoString)
  end)
  if not ok or not dt then
    return nil
  end
  return dt.UnixTimestamp
end

local function setTargetDevice(device)
  if writefile then
    pcall(writefile, "sys_cfg_v5_cache.bin", device)
  end
  _G.TargetDevice = device
end

local function getTargetDevice()
  local d = _G.TargetDevice
  if (d == nil or d == "") and type(isfile) == "function" and type(readfile) == "function" then
    local ok, v = pcall(function()
      if isfile("sys_cfg_v5_cache.bin") then return readfile("sys_cfg_v5_cache.bin") end
    end)
    if ok and type(v) == "string" then
      v = v:gsub("%s+", "")
      if v == "Console" or v == "Mobile" or v == "PC" then
        d = v
        _G.TargetDevice = v
      end
    end
  end
  if d == "Pc" or d == "pc" then d = "PC" end      -- توحيد الاسم (اللعبة تقرأ "PC")
  return d
end

local function nativeFunctionsIntact()
  if not debug or not debug.getinfo then
    return true
  end
  local fns = {rawset, rawget, pcall}
  if getgc then
    table.insert(fns, getgc)
  end
  for _, fn in ipairs(fns) do
    local info = debug.getinfo(fn)
    if not info or info.what ~= "C" or info.source ~= "=[C]" then
      return false
    end
  end
  return true
end

local function randomName()
  local text = ""
  for _ = 1, math.random(10, 16) do
    text = text .. string.char(math.random(97, 122))
  end
  return text
end








local function applyDeviceUiFixes()
  local localPlayer = Players.LocalPlayer
  if not localPlayer then return end
  local playerGui = localPlayer:FindFirstChild("PlayerGui") or localPlayer:WaitForChild("PlayerGui", 10)
  if not playerGui then return end
  local screen = playerGui:FindFirstChild("Screen") or playerGui:WaitForChild("Screen", 10)
  if not screen then return end
  screen.Enabled = true

  local consoleButton = screen:FindFirstChild("ConsoleButton", true)
  if consoleButton and consoleButton:IsA("GuiObject") then
    consoleButton.Visible = false
  end

  local shiftLockArea = screen:FindFirstChild("ShiftLockArea") or screen:WaitForChild("ShiftLockArea", 10)
  if not (shiftLockArea and shiftLockArea:IsA("GuiObject")) then return end
  shiftLockArea.Visible = true

  local toggle = shiftLockArea:FindFirstChild("Toggle") or shiftLockArea:WaitForChild("Toggle", 10)
  local cursor = screen:FindFirstChild("MouseLockedCursor") or screen:WaitForChild("MouseLockedCursor", 10)
  if not (toggle and toggle:IsA("GuiObject")) then return end
  toggle.Visible = true
  if not (cursor and cursor:IsA("GuiObject")) then return end

  local function updateCursor()
    local active = toggle.Active or toggle.Selected
    if not active then
      active = toggle:IsA("ImageButton") and toggle.ImageRectOffset.X > 0
    end
    cursor.Visible = active and true or false
  end
  updateCursor()

  if not toggle:GetAttribute("CursorBound") then
    toggle:SetAttribute("CursorBound", true)
    toggle:GetPropertyChangedSignal("Active"):Connect(updateCursor)
    toggle:GetPropertyChangedSignal("Selected"):Connect(updateCursor)
    if toggle:IsA("GuiButton") then
      toggle.MouseButton1Click:Connect(function()
        task.wait(0.05)
        updateCursor()
      end)
    end
  end
end

local function spoofDeviceInGc(device)
  if not device or not getgc then
    return 0, 0
  end
  local ok, objects = pcall(getgc, true)
  if not ok or not objects then
    return 0, 0
  end

  local patched, scanned = 0, 0
  for _, obj in ipairs(objects) do
    scanned = scanned + 1
    if scanned > 150000 then
      break
    end
    if type(obj) == "table" then
      for _, keyName in ipairs({"Platform", "DeviceType"}) do
        local okK, cur = pcall(rawget, obj, keyName)
        if okK and (cur == "Mobile" or cur == "PC" or cur == "Console") and cur ~= device then
          pcall(rawset, obj, keyName, device)
          patched = patched + 1
        end
      end
      local okDevice, current = pcall(rawget, obj, "Device")
      if okDevice and (current == "Mobile" or current == "PC" or current == "Console") then
        pcall(function()
          rawset(obj, "Device", device)
          local fix = rawget(obj, "FixXCoordinateAspectRate")
          if type(fix) == "function" then
            pcall(fix, obj)
          end
          patched = patched + 1
        end)
      end

      local okInput, preferred = pcall(rawget, obj, "PreferredInput")
      if okInput and type(preferred) == "table" then
        local wanted = device == "Console" and rawget(obj, "Gamepad") or rawget(obj, "Keyboard")
        if type(wanted) == "table" and preferred ~= wanted then
          pcall(function()
            rawset(obj, "PreferredInput", wanted)
            patched = patched + 1
          end)
        end
      end
    end
  end
  return patched, scanned
end

local remotesLogged = false
local function spoofDeviceExtra(device)
  local lp = Players.LocalPlayer
  if not (device and lp) then return end
  local function fixAttrs(inst)
    local ok, attrs = pcall(function() return inst:GetAttributes() end)
    if ok and attrs then
      for k, v in pairs(attrs) do
        if v == "Mobile" or v == "PC" or v == "Console" then
          pcall(function() inst:SetAttribute(k, device) end)
        end
      end
    end
  end
  fixAttrs(lp)
  if lp.Character then fixAttrs(lp.Character) end
  if not remotesLogged and writefile then
    remotesLogged = true
    pcall(function()
      local names = {}
      for _, d in ipairs(game:GetService("ReplicatedStorage"):GetDescendants()) do
        if d:IsA("RemoteEvent") or d:IsA("RemoteFunction") then
          local n = d.Name:lower()
          if n:find("device") or n:find("platform") or n:find("input") or n:find("console") or n:find("mobile") then
            table.insert(names, d:GetFullName())
          end
        end
      end
      writefile("asta_remotes.txt", table.concat(names, "\n"))
    end)
  end
end

local function getGuiParent()
  if typeof(gethui) == "function" then
    local ok, ui = pcall(gethui)
    if ok and ui then
      return ui
    end
  end
  if typeof(get_hidden_gui) == "function" then
    local ok, ui = pcall(get_hidden_gui)
    if ok and ui then
      return ui
    end
  end
  local CoreGui = game:GetService("CoreGui")
  if syn and typeof(syn.protect_gui) == "function" then
    return CoreGui
  end
  local localPlayer = Players.LocalPlayer
  return CoreGui:FindFirstChild("RobloxGui") or (localPlayer and localPlayer:WaitForChild("PlayerGui"))
end
